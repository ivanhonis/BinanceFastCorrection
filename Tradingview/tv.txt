round_down(number, decimals) =>
    factor = math.pow(10, decimals)
    math.floor(number * factor) / factor

rsrc = round_down(src, 8)

pine_ema(src, length) =>
    alpha = math.round(2 / (length + 1), 8)
    sum = 0.0
    if na(sum[1])
        sum := math.round(src, 8)
    else
        sum := math.round(alpha * src + (1 - alpha) * nz(sum[1]), 8)
    sum
